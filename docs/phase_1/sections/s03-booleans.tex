\subsection*{S3. Booleans}

A path holding JSON \texttt{true} and \texttt{false}.

\begin{lstlisting}[style=json]
{ "active": true }
\end{lstlisting}

\options
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{OCaml \texttt{bool}}. \selected
  \item \textbf{A generated two-constructor variant}, for uniformity with
        whatever S18 chooses for paths carrying several types.
\end{enumerate}

\decision{OCaml \texttt{bool}.}

\begin{lstlisting}[style=json]
{ "active": true }
{ "active": false }

  -> active : bool
\end{lstlisting}

\rationale{There is no choice to make here. A path that mixes booleans with
values of another type is not this scenario; it is S18.}
